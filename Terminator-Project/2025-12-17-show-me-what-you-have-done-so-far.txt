  Example 2: Safety Validator Integration

  Create new file: core/terminator_command_safety.py

  """
  Command Safety Validator for Terminator
  Adapted from external terminal_agent.py with enhancements
  """

  import shlex
  import os
  from typing import Tuple, Optional

  class TerminatorCommandSafety:
      """Enhanced command safety validation"""

      def __init__(self):
          # Safe commands that don't require confirmation
          self.safe_commands = {
              'ls', 'pwd', 'echo', 'date', 'whoami', 'uname',
              'ps', 'top', 'which', 'grep', 'find', 'cat',
              'git', 'python', 'python3', 'pip', 'pip3',
              'npm', 'node', 'curl', 'wget', 'ping'
          }

          # Patterns that should NEVER be executed
          self.blocked_patterns = [
              'rm -rf /',
              'rm -rf ~',
              'dd if=',
              ':(){ :|:& };:',  # Fork bomb
              'mkfs',
              'format',
              '> /dev/sd',
              'sudo rm -rf'
          ]

          # Patterns requiring explicit confirmation
          self.warning_patterns = [
              'rm -r',
              'sudo',
              'chmod 777',
              'chown',
              'kill -9'
          ]

      def validate_command(self, command: str) -> Tuple[str, Optional[str]]:
          """
          Validate command safety.
          
          Returns:
              Tuple of (status, message) where status is:
              - 'SAFE': Execute without confirmation
              - 'WARNING': Requires user confirmation
              - 'BLOCKED': Never execute
          """
          command_lower = command.lower()

          # Check blocked patterns
          for pattern in self.blocked_patterns:
              if pattern in command_lower:
                  return ('BLOCKED', f"Dangerous pattern detected: {pattern}")

          # Check warning patterns
          for pattern in self.warning_patterns:
              if pattern in command_lower:
                  return ('WARNING', f"Potentially dangerous: {pattern}")

          # Check if base command is safe
          try:
              base_command = shlex.split(command)[0]
              base_name = os.path.basename(base_command)

              if base_name in self.safe_commands:
                  return ('SAFE', None)
              else:
                  return ('WARNING', f"Unknown command: {base_name}")
          except:
              return ('BLOCKED', "Could not parse command")
